Quick Start

The easiest way — paste two lines into any HTML page:

<!-- Place this where you want the button -->
<div data-nlg-login></div>

<!-- Drop-in script (replace YOUR_CLIENT_ID and YOUR_CALLBACK_URL) -->
<script src="{{ base_url }}/oauth/widget.js
  ?client_id=YOUR_CLIENT_ID
  &redirect_uri=YOUR_CALLBACK_URL
  &scope=openid+profile+email
  &theme=auto"></script>

const CLIENT_ID     = process.env.NLG_CLIENT_ID;

  + `&redirect_uri=${encodeURIComponent(REDIRECT_URI)}`
  + `&response_type=code&scope=openid+profile+email`;

// 2. In your Express callback handler:
app.get('/auth/callback', async (req, res) => {
  const { data: tokens } = await axios.post(`${BASE}/oauth/token`,
    new URLSearchParams({
      grant_type:    'authorization_code',
      code:          req.query.code,
      redirect_uri:  REDIRECT_URI,
      client_id:     CLIENT_ID,
      client_secret: CLIENT_SECRET,
    })
  );

  // 3. Get user info:
  const { data: user } = await axios.get(`${BASE}/oauth/userinfo`, {
    headers: { Authorization: `Bearer ${tokens.access_token}` }
  });
  console.log(user);  // { sub: '1', name: '…', email: '…@netlifegy.com' }
});

For mobile apps and SPAs — no client secret needed, extra secure:

// JavaScript (works in browser or React Native)
import crypto from 'crypto';  // use expo-crypto on React Native

function base64url(buf) {
  return buf.toString('base64').replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
}

// 1. Generate PKCE pair
const verifier  = base64url(crypto.randomBytes(32));
const challenge = base64url(crypto.createHash('sha256').update(verifier).digest());

sessionStorage.setItem('pkce_verifier', verifier);

// 2. Redirect to authorization URL (no client_secret)
const authUrl = '{{ base_url }}/oauth/authorize'
  + '?client_id=YOUR_CLIENT_ID'
  + '&redirect_uri=' + encodeURIComponent('yourapp://callback')
  + '&response_type=code'
  + '&scope=openid+profile+email'
  + '&code_challenge=' + challenge
  + '&code_challenge_method=S256';

window.location.href = authUrl;

// 3. In your callback — exchange code WITHOUT client_secret
async function handleCallback(code) {
  const res = await fetch('{{ base_url }}/oauth/token', {
    method: 'POST',
    body: new URLSearchParams({
      grant_type:     'authorization_code',
      code:           code,
      redirect_uri:   'yourapp://callback',
      client_id:      'YOUR_CLIENT_ID',
      code_verifier:  sessionStorage.getItem('pkce_verifier'),
      // No client_secret needed!
    }),
  });
  const { access_token, refresh_token, id_token } = await res.json();
}
